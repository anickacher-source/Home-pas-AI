📦 ຈັດການສະຕັອກສິນຄ້າ

⚙️ ຕັ້ງຄ່າເຊື່ອມຕໍ່ຄລາວ (Supabase) — ເຮັດຄັ້ງດຽວ

ຕ້ອງການ Project URL ແລະ anon public key ຈາກ Supabase ຂອງທ່ານ (ເບິ່ງວິທີຕັ້ງຄ່າລຸ່ມສຸດ)

📜 ວິທີຕັ້ງຄ່າ Supabase (ເຮັດຄັ້ງດຽວ, ຟຣີ)
  1. ໄປ supabase.com → ສະໝັກບັນຊີຟຣີ → "New Project"
  2. ຕັ້ງຊື່ project ແລະລະຫັດຜ່ານຖານຂໍ້ມູນ (ຈື່ໄວ້), ລໍຖ້າ ~2 ນາທີໃຫ້ສ້າງສຳເລັດ
  3. ໄປ Project Settings → API → copy "Project URL" ແລະ "anon public" key ມາໃສ່ຊ່ອງເທິງນີ້
  4. ໄປ SQL Editor → ວາງໂຄ້ດນີ້ → ກົດ Run (ຄັ້ງດຽວ): 
    create table stock_items (
  id text not null, user_id uuid not null references auth.users(id) on delete cascade,
  data jsonb not null default '{}'::jsonb, deleted boolean not null default false,
  updated_at bigint not null default 0, primary key (user_id, id));
create table stock_history (
  id text not null, user_id uuid not null references auth.users(id) on delete cascade,
  data jsonb not null default '{}'::jsonb, deleted boolean not null default false,
  updated_at bigint not null default 0, primary key (user_id, id));
create table stock_biz_types (
  id text not null, user_id uuid not null references auth.users(id) on delete cascade,
  data jsonb not null default '{}'::jsonb, deleted boolean not null default false,
  updated_at bigint not null default 0, primary key (user_id, id));
create table stock_rate_history (
  id text not null, user_id uuid not null references auth.users(id) on delete cascade,
  data jsonb not null default '{}'::jsonb, deleted boolean not null default false,
  updated_at bigint not null default 0, primary key (user_id, id));
create table stock_settings (
  user_id uuid primary key references auth.users(id) on delete cascade,
  rates jsonb not null default '{}'::jsonb, updated_at bigint not null default 0);

alter table stock_items enable row level security;
alter table stock_history enable row level security;
alter table stock_biz_types enable row level security;
alter table stock_rate_history enable row level security;
alter table stock_settings enable row level security;

create policy "own rows" on stock_items for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
create policy "own rows" on stock_history for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
create policy "own rows" on stock_biz_types for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
create policy "own rows" on stock_rate_history for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
create policy "own rows" on stock_settings for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
  5. (ທາງເລືອກ — ຖ້າຕ້ອງການບັນຊີພະນັກງານ) run SQL ນີ້ເພີ່ມ ອີກຄັ້ງດຽວ ຫຼັງຈາກ SQL ຂ້າງເທິງ: 
    create table stock_members (
  member_id uuid primary key references auth.users(id) on delete cascade,
  shop_id uuid not null references auth.users(id) on delete cascade,
  email text not null);
create table stock_invites (
  shop_id uuid not null references auth.users(id) on delete cascade,
  email text not null, primary key (shop_id, email));
alter table stock_members enable row level security;
alter table stock_invites enable row level security;

create or replace function stock_can_access(p_shop uuid) returns boolean
language sql stable security definer set search_path = public as $$
  select auth.uid() = p_shop or exists (
    select 1 from stock_members where member_id = auth.uid() and shop_id = p_shop) $$;

drop policy if exists "own rows" on stock_items;
drop policy if exists "own rows" on stock_history;
drop policy if exists "own rows" on stock_biz_types;
drop policy if exists "own rows" on stock_rate_history;
drop policy if exists "own rows" on stock_settings;
create policy "shop rows" on stock_items for all using (stock_can_access(user_id)) with check (stock_can_access(user_id));
create policy "shop rows" on stock_history for all using (stock_can_access(user_id)) with check (stock_can_access(user_id));
create policy "shop rows" on stock_biz_types for all using (stock_can_access(user_id)) with check (stock_can_access(user_id));
create policy "shop rows" on stock_rate_history for all using (stock_can_access(user_id)) with check (stock_can_access(user_id));
create policy "shop rows" on stock_settings for all using (stock_can_access(user_id)) with check (stock_can_access(user_id));

create policy "see members" on stock_members for select using (member_id = auth.uid() or shop_id = auth.uid());
create policy "owner removes members" on stock_members for delete using (shop_id = auth.uid());
create policy "owner manages invites" on stock_invites for all using (shop_id = auth.uid()) with check (shop_id = auth.uid());

create or replace function stock_my_shop() returns uuid
language plpgsql security definer set search_path = public as $$
declare v_shop uuid; v_email text := lower(auth.jwt() ->> 'email');
begin
  select shop_id into v_shop from stock_members where member_id = auth.uid();
  if v_shop is not null then return v_shop; end if;
  select shop_id into v_shop from stock_invites where lower(email) = v_email limit 1;
  if v_shop is null then return null; end if;
  insert into stock_members (member_id, shop_id, email) values (auth.uid(), v_shop, v_email);
  delete from stock_invites where lower(email) = v_email;
  return v_shop;
end $$;
  6. ກັບມາໜ້ານີ້ ວາງ URL+key ແລ້ວກົດ "ເຊື່ອມຕໍ່"

🔑 ກູ້ລະຫັດຜ່ານ

ໃສ່ອີເມວທີ່ສະໝັກໄວ້ — ລະບົບຈະສົ່ງລິ້ງກູ້ລະຫັດຜ່ານໄປໃຫ້ (ໄປຕັ້ງລະຫັດຜ່ານໃໝ່ຈາກລິ້ງນັ້ນ)

⚠️ ໃຊ້ໄດ້ດີຖ້າເປີດຜ່ານລິ້ງເວັບ (https://...). ຖ້າເປີດຈາກໄຟລ໌ໃນຄອມໂດຍກົງ ລິ້ງກູ້ລະຫັດອາດບໍ່ເຮັດວຽກ (Supabase ບໍ່ຍອມຮັບລິ້ງແບບ file://) — ກໍລະນີນີ້ໃຫ້ເຈົ້າຂອງບັນຊີ Supabase ໄປຕັ້ງລະຫັດຜ່ານໃໝ່ໃຫ້ຜ່ານ Dashboard → Authentication → Users ແທນ

🔑 ຕັ້ງລະຫັດຜ່ານໃໝ່

📦 ຈັດການສະຕັອກສິນຄ້າ

ຕ້ອງໃຊ້ Chrome ຫຼື Edge ຢູ່ຄອມ. ບາງ browser ຈະຖາມອະນຸຍາດໂຟນເດີໃໝ່ທຸກຄັ້ງທີ່ເປີດໄຟລ໌ນີ້ — ເປັນເລື່ອງປົກກະຕິ, ກົດອະນຸຍາດໄດ້ເລີຍ.

⚠️ ບໍ່ໄດ້ກົດ Export Excel ມາດົນແລ້ວ (ຫຼືຍັງບໍ່ເຄີຍເລີຍ) — ຂໍ້ມູນຍັງຢູ່ໃນ browser ນີ້ຢ່າງດຽວ, ແນະນຳໃຫ້ກົດ "Export Excel" ເກັບໄວ້ເປັນສຳເນົາ
👥 ພະນັກງານ (ເຊີນ/ລົບ)

ໃສ່ອີເມວພະນັກງານ ແລ້ວກົດເຊີນ — ພະນັກງານສະໝັກບັນຊີດ້ວຍອີເມວນັ້ນ ແລ້ວຈະເຂົ້າຮ້ານນີ້ອັດຕະໂນມັດ. ພະນັກງານຂາຍ/ຮັບເຂົ້າໄດ້ ແຕ່ບໍ່ເຫັນຕົ້ນທຶນ/ກຳໄລ/ລາຍງານ ແລະລົບ/ແກ້ໄຂບໍ່ໄດ້ (ເປັນການເຊື່ອງໃນໜ້າຈໍເທົ່ານັ້ນ).
